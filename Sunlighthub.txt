local function vflyGetCharacterData()
    local char = LocalPlayer.Character
    if not char or not char.Parent then return nil, nil, nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hum.Parent or not root or not root.Parent then return nil, nil, nil end
    return char, hum, root
end

local function vflyStop()
    VehicleFlySettings.Enabled = false
    if VehicleFlySettings.HeartbeatConn then
        pcall(function() VehicleFlySettings.HeartbeatConn:Disconnect() end)
        VehicleFlySettings.HeartbeatConn = nil
    end
    if VehicleFlySettings.DiedConn then
        pcall(function() VehicleFlySettings.DiedConn:Disconnect() end)
        VehicleFlySettings.DiedConn = nil
    end
    local root = VehicleFlySettings.Root
    if root and root.Parent then
        for _, name in ipairs({"__VFlyBV", "__VFlyBG"}) do
            local obj = root:FindFirstChild(name)
            if obj then pcall(function() obj:Destroy() end) end
        end
    end
    local hum = VehicleFlySettings.Humanoid
    if hum and hum.Parent then
        pcall(function()
            hum.PlatformStand = false
            hum.AutoRotate = VehicleFlySettings.OriginalAutoRotate
            for state, enabled in pairs(VehicleFlySettings.OriginalStates) do
                hum:SetStateEnabled(state, enabled)
            end
        end)
    end
    local char = LocalPlayer.Character
    if char and char:FindFirstChild("Animate") then
        pcall(function() char.Animate.Disabled = false end)
    end
    VehicleFlySettings.Humanoid = nil
    VehicleFlySettings.Root = nil
    VehicleFlySettings.BV = nil
    VehicleFlySettings.BG = nil
    VehicleFlySettings.UpHeld = false
    VehicleFlySettings.DownHeld = false
    VehicleFlySettings.ForwardHeld = false
    VehicleFlySettings.BackHeld = false
    VehicleFlySettings.LeftHeld = false
    VehicleFlySettings.RightHeld = false
    VehicleFlySettings.OriginalStates = {}
end

local function vflyStart()
    vflyStop()
    local char, hum, root = vflyGetCharacterData()
    if not char or not hum or not root then return end
    VehicleFlySettings.Enabled = true
    VehicleFlySettings.Humanoid = hum
    VehicleFlySettings.Root = root
    VehicleFlySettings.OriginalAutoRotate = hum.AutoRotate

    local states = {
        Enum.HumanoidStateType.FallingDown,
        Enum.HumanoidStateType.Ragdoll,
        Enum.HumanoidStateType.Physics,
        Enum.HumanoidStateType.PlatformStanding,
        Enum.HumanoidStateType.Freefall,
    }
    for _, state in ipairs(states) do
        VehicleFlySettings.OriginalStates[state] = hum:GetStateEnabled(state)
        pcall(function() hum:SetStateEnabled(state, false) end)
    end
    if char:FindFirstChild("Animate") then char.Animate.Disabled = true end
    hum.AutoRotate = false

    local bv = Instance.new("BodyVelocity")
    bv.Name = "__VFlyBV"
    bv.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    bv.Velocity = Vector3.zero
    bv.P = 1250
    bv.Parent = root
    VehicleFlySettings.BV = bv

    local bg = Instance.new("BodyGyro")
    bg.Name = "__VFlyBG"
    bg.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    bg.P = 10000
    bg.D = 500
    bg.CFrame = root.CFrame
    bg.Parent = root
    VehicleFlySettings.BG = bg

    pcall(function() hum:ChangeState(Enum.HumanoidStateType.Swimming) end)
    VehicleFlySettings.DiedConn = hum.Died:Connect(function() vflyStop() end)

    VehicleFlySettings.HeartbeatConn = RunService.Heartbeat:Connect(function(dt)
        if not VehicleFlySettings.Enabled then return end
        if not root or not root.Parent then vflyStop() return end
        if not hum or not hum.Parent then vflyStop() return end
        local camera = Workspace.CurrentCamera
        if not camera then return end

        local camLook = camera.CFrame.LookVector
        local camRight = camera.CFrame.RightVector
        local forwardDir = Vector3.new(camLook.X, 0, camLook.Z)
        if forwardDir.Magnitude > 0.001 then forwardDir = forwardDir.Unit end
        local rightDir = Vector3.new(camRight.X, 0, camRight.Z)
        if rightDir.Magnitude > 0.001 then rightDir = rightDir.Unit end

        local analogDir = hum.MoveDirection
        local buttonDir = Vector3.zero
        if VehicleFlySettings.ForwardHeld then buttonDir += forwardDir end
        if VehicleFlySettings.BackHeld then buttonDir -= forwardDir end
        if VehicleFlySettings.RightHeld then buttonDir += rightDir end
        if VehicleFlySettings.LeftHeld then buttonDir -= rightDir end

        local finalMoveDir
        if analogDir.Magnitude > 0.05 then
            finalMoveDir = analogDir
        else
            finalMoveDir = buttonDir
        end

        local horizontalVel = Vector3.zero
        if finalMoveDir.Magnitude > 0.05 then
            local forwardAmount = finalMoveDir:Dot(forwardDir)
            local rightAmount = finalMoveDir:Dot(rightDir)
            horizontalVel = (forwardDir * forwardAmount + rightDir * rightAmount)
            if horizontalVel.Magnitude > 0.001 then
                horizontalVel = horizontalVel.Unit * VehicleFlySettings.Speed * math.min(finalMoveDir.Magnitude, 1)
            end
        end

        local verticalVel = 0
        if VehicleFlySettings.UpHeld then
            verticalVel = VehicleFlySettings.UpSpeed
        elseif VehicleFlySettings.DownHeld then
            verticalVel = -VehicleFlySettings.DownSpeed
        end

        local targetVel = Vector3.new(horizontalVel.X, verticalVel, horizontalVel.Z)
        local alpha = 1 - math.exp(-(1 / VehicleFlySettings.Smoothness) * dt)
        bv.Velocity = bv.Velocity:Lerp(targetVel, alpha)

        local lookDir = Vector3.new(horizontalVel.X, 0, horizontalVel.Z)
        if lookDir.Magnitude > 0.001 then
            bg.CFrame = CFrame.lookAt(root.Position, root.Position + lookDir.Unit, Vector3.yAxis)
        else
            local camForward = Vector3.new(camLook.X, 0, camLook.Z)
            if camForward.Magnitude > 0.001 then
                bg.CFrame = CFrame.lookAt(root.Position, root.Position + camForward.Unit, Vector3.yAxis)
            end
        end

        if root.AssemblyLinearVelocity.Y < -50 and verticalVel >= 0 then
            bv.Velocity = Vector3.new(bv.Velocity.X, 0, bv.Velocity.Z)
        end
    end)
end

local function vflyBuildGui()
    if VehicleFlySettings.Gui and VehicleFlySettings.Gui.Parent then
        VehicleFlySettings.Gui.Enabled = true
        return

    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "SunlightVehicleFly"

    pcall(function()
        if syn and syn.protect_gui then syn.protect_gui(screenGui) end
    end)
    screenGui.Parent = LocalPlayer:WaitForChild("PlayerGui")

    local function makeDraggable(frame, handle)
        local dragging = false
        local dragStart, startPos
        handle.InputBegan:Connect(function(input)
            if VehicleFlySettings.Locked then return end
            if input.UserInputType == Enum.UserInputType.MouseButton1 
                or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = frame.Position
            end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if VehicleFlySettings.Locked then dragging = false; return end
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement 
                or input.UserInputType == Enum.UserInputType.Touch) then
                local delta = input.Position - dragStart
                frame.Position = UDim2.new(
                    startPos.X.Scale, startPos.X.Offset + delta.X,
                    startPos.Y.Scale, startPos.Y.Offset + delta.Y
                )
            end
        end)
        UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 
                or input.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end)
    end

    local function bindHold(button, onTrue, onFalse)
        local isHeld = false
        local function press()
            if not isHeld then isHeld = true; onTrue() end
        end
        local function release()
            if isHeld then isHeld = false; onFalse() end
        end
        button.MouseButton1Down:Connect(press)
        button.MouseButton1Up:Connect(release)
        button.MouseLeave:Connect(release)
        button.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch 
                or input.UserInputType == Enum.UserInputType.MouseButton1 then
                press()
            end
        end)
        button.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch 
                or input.UserInputType == Enum.UserInputType.MouseButton1 then
                release()
            end
        end)
    end

        local function canRingPart(part, targetRoot)
                if not part or not part:IsA("BasePart") or not part.Parent or part.Anchored
                        or part:IsA("Seat") or part:IsA("VehicleSeat") or part.AssemblyRootPart ~= part then
                        return false

                if LocalPlayer.Character and part:IsDescendantOf(LocalPlayer.Character) then return false end
                if VehicleSettings.CurrentModel and VehicleSettings.CurrentModel.Parent and part:IsDescendantOf(VehicleSettings.CurrentModel) then return false end
                if hasPartRingDisallowedAncestor(part) then return false end
                local maximumAssemblyMass = math.clamp(tonumber(PartRingSettings.MaximumAssemblyMass) or 5000000, 1, 10000000)
                local maximumAssemblySize = math.clamp(tonumber(PartRingSettings.MaximumAssemblySize) or 300, 1, 10000)
                if part.AssemblyMass > maximumAssemblyMass or part.Size.Magnitude > maximumAssemblySize then return false end
                return (part.Position - targetRoot.Position).Magnitude <= getPartRingSearchRadius()
        end

        local function getPartRingState(root)
                local state = PartRingSettings.PartStates[root]
                if state then return state end
                state = {
                        CollisionParts = {}, Angle = nil, FailCount = 0, PrevPos = nil, LastSlot = nil,
                        RadiusScale = 0.3 + math.random() * 1.4,
                        HeightOffset = (math.random() - 0.5) * 10,
                        SpeedScale = 0.45 + math.random() * 1.3,
                        Direction = math.random() < 0.5 and -1 or 1,
                        WobbleAmplitude = 1 + math.random() * 5,
                        WobbleFrequency = 0.8 + math.random() * 2.6,
                        WobblePhase = math.random() * TAU,
                        SpinAxis = Vector3.new((math.random() - 0.5) * 0.8, 1, (math.random() - 0.5) * 0.8).Unit,
                        SpinScale = 0.55 + math.random() * 0.9,
                        SpinSign = math.random() < 0.5 and -1 or 1

                return state

        local function restorePartRingPart(root)
                local state = PartRingSettings.PartStates[root]
                local controlledParts = state and state.CollisionParts or {root}
                removeLegacyPartRingConstraints(root)
                local releasePower = math.clamp(tonumber(PartRingSettings.ReleasePower) or 1, 0, 5)
                for _, part in ipairs(controlledParts) do
                        local originalCollision = PartRingSettings.CollisionStates[part]
                        if part and part.Parent then
                                pcall(function()
                                        if originalCollision ~= nil then part.CanCollide = originalCollision end
                                end)
                        end
                        PartRingSettings.CollisionStates[part] = nil
                end
                if root and root.Parent then
                        pcall(function()
                                if releasePower <= 0.001 then
                                        root.AssemblyLinearVelocity = Vector3.zero
                                        root.AssemblyAngularVelocity = Vector3.zero
                                else
                                        local linearVelocity = root.AssemblyLinearVelocity
                                        local angularVelocity = root.AssemblyAngularVelocity
                                        local verticalBoost = (releasePower - 1) * 25
                                        root.AssemblyLinearVelocity = linearVelocity * releasePower + Vector3.new(0, verticalBoost, 0)
                                        root.AssemblyAngularVelocity = angularVelocity * math.min(releasePower, 2)
                                end
                        end)
                end
                PartRingSettings.PartStates[root] = nil
        end

        local function preparePartForRing(root)
                removeLegacyPartRingConstraints(root)
                local state = getPartRingState(root)
                local known = {}
                for _, part in ipairs(state.CollisionParts) do known[part] = true end
                for _, part in ipairs(getPartRingAssemblyParts(root)) do
                        if not known[part] then
                                known[part] = true
                                table.insert(state.CollisionParts, part)
                        end
                        if PartRingSettings.CollisionStates[part] == nil then
                                PartRingSettings.CollisionStates[part] = part.CanCollide
                        end
                        pcall(function() part.CanCollide = false end)
                end
        end

        local function isPartRingViable(part, targetRoot, distanceLimit)
                if not part or not part.Parent or part.AssemblyRootPart ~= part then return false end
                local offset = part.Position - targetRoot.Position
                return offset.Magnitude <= distanceLimit
        end

        local function applyRingPlacement(part, targetNow, targetNext, tangentVelocity, stepTime, pullGain, carry, angularVelocity)
                part.CanCollide = false
                local currentPosition = part.Position
                local residual = targetNext - currentPosition - tangentVelocity * stepTime
                local correction = residual / stepTime * pullGain + Vector3.new(0, 0.5 * Workspace.Gravity * stepTime, 0)
                local velocity = tangentVelocity + correction + carry
                part.AssemblyLinearVelocity = velocity
                part.AssemblyAngularVelocity = angularVelocity
                return velocity, currentPosition
        end

                local total = #parts
                if total > 0 then
                        local moving = tonumber(PartRingSettings.MovingCount) or 0
                        local label
                        if moving >= total then
                                label = "Chaos: " .. tostring(total) .. " parts swarming"
                        else
                                label = "Chaos: " .. tostring(moving) .. " of " .. tostring(total) .. " parts swarming (rest waiting on physics)"
                        end
                        updatePartRingStatus(label)
                else
                        updatePartRingStatus("No eligible loose parts")
                end
        end

        PartRing.refresh = function()
                local targetRoot = PartRing.getTargetRoot()
                if not targetRoot or not targetRoot.Parent then
                        clearPartRingParts()
                        updatePartRingStatus("Target unavailable")
                        return
                end
                local now = os.clock()
                local overlap = OverlapParams.new()
                overlap.FilterType = Enum.RaycastFilterType.Exclude
                local filter = {}
                if LocalPlayer.Character then table.insert(filter, LocalPlayer.Character) end
                local targetPlayer = PartRingSettings.TargetUserId and Players:GetPlayerByUserId(PartRingSettings.TargetUserId) or LocalPlayer
                if targetPlayer and targetPlayer.Character and targetPlayer.Character ~= LocalPlayer.Character then
                        table.insert(filter, targetPlayer.Character)
                end
                overlap.FilterDescendantsInstances = filter
                overlap.MaxParts = 0
                local nearby = {}
                pcall(function()
                        nearby = Workspace:GetPartBoundsInRadius(targetRoot.Position, getPartRingSearchRadius(), overlap)
                end)
                local candidates = {}
                local candidateSet = {}
                for _, part in ipairs(nearby) do
                        local cooldown = PartRingSettings.Cooldowns[part]
                        if canRingPart(part, targetRoot) and not candidateSet[part] and not (cooldown and cooldown > now) then
                                candidateSet[part] = true
                                table.insert(candidates, part)
                        end
                end
                table.sort(candidates, function(first, second)
                        return (first.Position - targetRoot.Position).Magnitude < (second.Position - targetRoot.Position).Magnitude
                end)
                local nextParts = {}
                local selected = {}
                for _, part in ipairs(PartRingSettings.Parts) do
                        if candidateSet[part] then
                                selected[part] = true
                                table.insert(nextParts, part)
                        else
                                restorePartRingPart(part)
                        end
                end
                for _, part in ipairs(candidates) do
                        if not selected[part] then
                                selected[part] = true
                                table.insert(nextParts, part)
                        end
                end
                PartRingSettings.Parts = nextParts
                for _, part in ipairs(nextParts) do preparePartForRing(part) end
                PartRingSettings.NextRescan = os.clock() + PartRingSettings.RescanInterval
                composePartRingStatus()
        end

        PartRing.stop = function()
                disconnect(PartRingSettings.Connection)
                PartRingSettings.Connection = nil
                clearPartRingParts()
                PartRingSettings.NextRescan = 0
                updatePartRingStatus("Chaos off")
        end

        local function updatePartRing(deltaTime)
                if not running or not PartRingSettings.Enabled then return end
                local targetRoot = PartRing.getTargetRoot()
                if not targetRoot or not targetRoot.Parent then
                        if #PartRingSettings.Parts > 0 then
                                clearPartRingParts()
                                updatePartRingStatus("Target unavailable")
                        end
                        return
                end
                local now = os.clock()
                if now >= PartRingSettings.NextRescan then PartRing.refresh() end
                local parts = PartRingSettings.Parts
                local searchLimit = getPartRingSearchRadius() * 1.02
                local removedAny = false
                for index = #parts, 1, -1 do
                        local part = parts[index]
                        local state = PartRingSettings.PartStates[part]
                        if not part or not part.Parent then
                                restorePartRingPart(part)
                                table.remove(parts, index)
                                removedAny = true
                        else
                                local failedTooOften = state ~= nil and (state.FailCount or 0) >= 30
                                if failedTooOften or not isPartRingViable(part, targetRoot, searchLimit) then
                                        restorePartRingPart(part)
                                        if failedTooOften then PartRingSettings.Cooldowns[part] = now + 6 end
                                        table.remove(parts, index)
                                        removedAny = true
                                end
                        end
                end
                if removedAny then composePartRingStatus() end
                local count = #parts
                if count == 0 then return end
                local radius = math.clamp(tonumber(PartRingSettings.Radius) or 12, 3, 150)
                local height = math.clamp(tonumber(PartRingSettings.Height) or 4, -10, 25)
                local spinSpeed = math.clamp(tonumber(PartRingSettings.Speed) or 3.5, 0, 20)
                local rotationSpeed = math.clamp(tonumber(PartRingSettings.RotationSpeed) or 30, 0, 300)
                local pullStrength = math.clamp(tonumber(PartRingSettings.PullStrength) or 60, 1, 100)
                local stepTime = math.clamp(tonumber(deltaTime) or 0, 0, 0.1)
                if stepTime < 0.0001 then return end
                local center = targetRoot.Position
                if center.X ~= center.X or center.Y ~= center.Y or center.Z ~= center.Z then return end
                local carry = targetRoot.AssemblyLinearVelocity * 0.5
                if carry.X ~= carry.X or carry.Y ~= carry.Y or carry.Z ~= carry.Z then
                        carry = Vector3.zero
                elseif carry.Magnitude > 250 then
                        carry = carry.Unit * 250
                end
                local pullGain = 0.5 + pullStrength * 0.014
                local slotTolerance = math.max(2.5, radius * 0.25)
                local movingCount = 0
                for index = 1, count do
                        local part = parts[index]
                        local state = PartRingSettings.PartStates[part]
                        if not state then state = getPartRingState(part) end
                        local currentPosition = part.Position
                        if state.Angle == nil then
                                local offsetX = currentPosition.X - center.X
                                local offsetZ = currentPosition.Z - center.Z
                                if offsetX ~= offsetX or offsetZ ~= offsetZ then
                                        state.Angle = 0
                                else
                                        state.Angle = math.atan2(offsetZ, offsetX) % TAU
                                end
                        end
                        local angularSpeed = spinSpeed * state.SpeedScale * state.Direction
                        local angle = state.Angle
                        local nextAngle = angle + angularSpeed * stepTime
                        state.Angle = nextAngle % TAU
                        local wobbleNow = math.sin(now * state.WobbleFrequency + state.WobblePhase) * state.WobbleAmplitude
                        local wobbleNext = math.sin((now + stepTime) * state.WobbleFrequency + state.WobblePhase) * state.WobbleAmplitude
                        local verticalNow = math.cos(now * state.WobbleFrequency * 0.7 + state.WobblePhase * 1.3) * state.WobbleAmplitude * 0.6
                        local verticalNext = math.cos((now + stepTime) * state.WobbleFrequency * 0.7 + state.WobblePhase * 1.3) * state.WobbleAmplitude * 0.6
                        local radiusNow = math.max(2, radius * state.RadiusScale + wobbleNow)
                        local radiusNext = math.max(2, radius * state.RadiusScale + wobbleNext)
                        local targetNow = center + Vector3.new(math.cos(angle) * radiusNow, height + state.HeightOffset + verticalNow, math.sin(angle) * radiusNow)
                        local targetNext = center + Vector3.new(math.cos(nextAngle) * radiusNext, height + state.HeightOffset + verticalNext, math.sin(nextAngle) * radiusNext)
                        local tangentSpeed = radiusNow * angularSpeed
                        local tangentVelocity = Vector3.new(-math.sin(angle) * tangentSpeed, 0, math.cos(angle) * tangentSpeed)
                        local angularVelocity = state.SpinAxis * (rotationSpeed * state.SpinScale * state.SpinSign)
                        if state.PrevPos then
                                local moved = (currentPosition - state.PrevPos).Magnitude
                                if moved ~= moved then moved = 0 end
                                if moved > 0.05 or (state.LastSlot and (currentPosition - state.LastSlot).Magnitude <= slotTolerance) then
                                        movingCount = movingCount + 1
                                end
                        end
                        local placed, _, preWritePosition = pcall(applyRingPlacement, part, targetNow, targetNext, tangentVelocity, stepTime, pullGain, carry, angularVelocity)
                        state.PrevPos = preWritePosition
                        state.LastSlot = targetNow
                        if placed then state.FailCount = 0 else state.FailCount = state.FailCount + 1 end
                end
                PartRingSettings.MovingCount = movingCount
        end

        PartRing.setEnabled = function(value)
                local enabled = value and true or false
                if not enabled then PartRing.setPickerEnabled(false) end
                PartRing.stop()
                PartRingSettings.Enabled = enabled
                if type(PartRing.OnEnabledChanged) == "function" then
                        pcall(PartRing.OnEnabledChanged, enabled)
                end
                if not enabled then return end
                PartRing.refresh()
                PartRingSettings.Connection = RunService.Heartbeat:Connect(
                        function(stepTime)
                                local ok, err = pcall(updatePartRing, stepTime)
                                if ok then
                                        PartRingSettings.ErrorCount = 0
                                        return
                                end
                                PartRingSettings.ErrorCount = PartRingSettings.ErrorCount + 1
                                if PartRingSettings.ErrorCount == 5 then
                                        updatePartRingStatus("Engine error: " .. tostring(err):sub(1, 80))
                                end
                        end
                )
        end
end

local function bpvAnimate(state, direction)
    if state.tweens then
        for _, t in ipairs(state.tweens) do
            pcall(function() t:Cancel() end)
        end
    end
    state.tweens = {}

    local transparencyGoal, sizeGoal
    if direction == "in" then
        transparencyGoal, sizeGoal = 0.2, UDim2.fromScale(1, 1)
    else
        transparencyGoal, sizeGoal = 1, UDim2.fromScale(0.8, 0.8)
    end

    local tweenService = game:GetService("TweenService")
    local transparencyTween = tweenService:Create(
        state.main,
        TweenInfo.new(BackpackViewerSettings.AnimTime, Enum.EasingStyle.Quint),
        { BackgroundTransparency = transparencyGoal }
    )
    local sizeTween = tweenService:Create(
        state.root,
        TweenInfo.new(BackpackViewerSettings.AnimTime, Enum.EasingStyle.Quint),
        { Size = sizeGoal }
    )

    table.insert(state.tweens, transparencyTween)
    table.insert(state.tweens, sizeTween)

    transparencyTween:Play()
    sizeTween:Play()

    return sizeTween
end

local function bpvUpdateUI(player, char)
    if not BackpackViewerSettings.Enabled then return end
    local state = BackpackViewerSettings.UIs[player]
    if not (state and state.gui) then return end

    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then
        local health = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
        local fill = state.main.HealthBar.Fill
        fill.Size = UDim2.fromScale(health, 1)
        fill.BackgroundColor3 = Color3.fromHSV(0.33 * health, 0.7, 0.8)
    end

    local pData = BPV_CONFIG.platforms[bpvGetPlayerPlatform(player)]
    local pIcon = state.main.PlatformIcon
    pIcon.Image = pData[1]
    pIcon.ImageColor3 = pData[2]

    local scroller = state.main.Backpack
    local tooltip = scroller.Tooltip
    for _, child in ipairs(scroller:GetChildren()) do
        if child:IsA("ImageButton") then child:Destroy() end
    end

    for _, tool in ipairs(player.Backpack:GetChildren()) do
        if tool:IsA("Tool") then
            local icon = Instance.new("ImageButton", scroller)
            icon.Size = UDim2.fromScale(1, 1)
            icon.BackgroundTransparency = 1
            icon.Image = tool.TextureId
            icon.MouseEnter:Connect(function()
                tooltip.Text = tool.Name
                tooltip.Visible = true
                tooltip.Parent = icon
            end)
            icon.MouseLeave:Connect(function()
                tooltip.Visible = false
                tooltip.Parent = scroller
            end)
            icon.MouseButton1Click:Connect(function()
                pcall(function()
                    tool:Clone().Parent = LocalPlayer.Backpack
                end)
            end)
        end
    end
end

local function bpvDestroyUI(player)
    local state = BackpackViewerSettings.UIs[player]
    if not state then return end
    if state.gui then
        pcall(function() state.gui:Destroy() end)
    end
    BackpackViewerSettings.UIs[player] = nil
end

local function bpvClearAll()
    for player, _ in pairs(BackpackViewerSettings.UIs) do
        bpvDestroyUI(player)
    end
    BackpackViewerSettings.UIs = {}
    BackpackViewerSettings.ActiveScroller = nil
end

local function setBackpackViewerEnabled(value)
    BackpackViewerSettings.Enabled = value and true or false

    -- ปิด connections เก่า
    for _, c in ipairs(BackpackViewerSettings.Connections) do
        pcall(function() c:Disconnect() end)
    end
    BackpackViewerSettings.Connections = {}

    if BackpackViewerSettings.HeartbeatConn then
        pcall(function() BackpackViewerSettings.HeartbeatConn:Disconnect() end)
        BackpackViewerSettings.HeartbeatConn = nil
    end

    if not BackpackViewerSettings.Enabled then
        bpvClearAll()
        return
    end

    -- Main loop
    BackpackViewerSettings.HeartbeatConn = RunService.Heartbeat:Connect(function()
        if not BackpackViewerSettings.Enabled then return end
        local localChar = LocalPlayer.Character
        if not (localChar and localChar.PrimaryPart) then return end
        local localPos = localChar.PrimaryPart.Position

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                local state = BackpackViewerSettings.UIs[player] or bpvCreateElements(player)
                local char = player.Character

                if char and char.PrimaryPart and char:FindFirstChild("Head") then
                    local dist = (localPos - char.PrimaryPart.Position).Magnitude

                    if dist <= BackpackViewerSettings.Distance and state.currentState == "hidden" then
                        state.currentState = "visible"
                        state.gui.Adornee = char.Head
                        state.gui.Parent = char.Head
                        bpvAnimate(state, "in")
                    elseif dist > BackpackViewerSettings.Distance and state.currentState == "visible" then
                        state.currentState = "hidden"
                        local tween = bpvAnimate(state, "out")
                        tween.Completed:Wait()
                        if state.currentState == "hidden" then state.gui.Parent = nil end
                    end

                    if state.currentState == "visible" then bpvUpdateUI(player, char) end
                elseif state.currentState == "visible" then
                    state.currentState = "hidden"
                    state.gui.Parent = nil
                end
            end
        end
    end)

    -- Gamepad scroll
    local scrollConn = UserInputService.InputChanged:Connect(function(input)
        if not BackpackViewerSettings.Enabled or not BackpackViewerSettings.ActiveScroller then return end
        if input.UserInputType == Enum.UserInputType.Gamepad1 and input.KeyCode == Enum.KeyCode.Gamepad1_Thumbstick2 then
            local scroller = BackpackViewerSettings.ActiveScroller
            scroller.CanvasPosition = scroller.CanvasPosition - Vector2.new(0, input.Position.Y * BackpackViewerSettings.ScrollSpeed)
        end
    end)
    table.insert(BackpackViewerSettings.Connections, scrollConn)

    -- Player leaving
    local leaveConn = Players.PlayerRemoving:Connect(function(player)
        bpvDestroyUI(player)
    end)
    table.insert(BackpackViewerSettings.Connections, leaveConn)
end

local function clearSpeedWatcher()
        disconnect(MovementSettings.WatchConnection)
        MovementSettings.WatchConnection = nil
end

local function applySpeed()
        local humanoid = MovementSettings.Humanoid
        if not running or not MovementSettings.Enabled or not humanoid or not humanoid.Parent then return end
        if humanoid.WalkSpeed == MovementSettings.Speed then return end
        MovementSettings.Updating = true
        humanoid.WalkSpeed = MovementSettings.Speed
        MovementSettings.Updating = false
end

local function watchSpeed()
        clearSpeedWatcher()
        local humanoid = MovementSettings.Humanoid
        if not humanoid or not humanoid.Parent then return end
        MovementSettings.WatchConnection = humanoid:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
                if running and MovementSettings.Enabled and not MovementSettings.Updating and humanoid.WalkSpeed ~= MovementSettings.Speed then
                        task.defer(applySpeed)
                end
        end)
end

-- ===== JUMP HACK =====
local function clearJumpWatcher()
        disconnect(MovementSettings.JumpWatchConnection)
        MovementSettings.JumpWatchConnection = nil
end

local function applyJump()
        local humanoid = MovementSettings.Humanoid
        if not running or not MovementSettings.JumpEnabled or not humanoid or not humanoid.Parent then return end
        if humanoid.JumpPower == MovementSettings.JumpPower then return end
        MovementSettings.JumpUpdating = true
        humanoid.UseJumpPower = true
        humanoid.JumpPower = MovementSettings.JumpPower
        MovementSettings.JumpUpdating = false
end

local function watchJump()
        clearJumpWatcher()
        local humanoid = MovementSettings.Humanoid
        if not humanoid or not humanoid.Parent then return end
        MovementSettings.JumpWatchConnection = humanoid:GetPropertyChangedSignal("JumpPower"):Connect(function()
                if running and MovementSettings.JumpEnabled and not MovementSettings.JumpUpdating and humanoid.JumpPower ~= MovementSettings.JumpPower then
                        task.defer(applyJump)
                end
        end)
end

local function setJumpEnabled(value)
        MovementSettings.JumpEnabled = value and true or false
        local humanoid = MovementSettings.Humanoid
        if not MovementSettings.JumpEnabled then
                clearJumpWatcher()
                if humanoid and humanoid.Parent and MovementSettings.OriginalJumpPower ~= nil then
                        MovementSettings.JumpUpdating = true
                        humanoid.JumpPower = MovementSettings.OriginalJumpPower
                        MovementSettings.JumpUpdating = false
                end
                MovementSettings.OriginalJumpPower = nil
                return
        end
        if not humanoid or not humanoid.Parent then
                bindMovementCharacter(LocalPlayer.Character)
                humanoid = MovementSettings.Humanoid
        end
        if humanoid and humanoid.Parent then
                MovementSettings.OriginalJumpPower = humanoid.JumpPower
                applyJump()
                watchJump()
        end
end

        if not running or not MovementSettings.Noclip then return end
        local character = LocalPlayer.Character

                        if MovementSettings.CollisionStates[object] == nil then
                                MovementSettings.CollisionStates[object] = object.CanCollide

                                object.CanCollide = false
                        end
                end
        end
end

local function clearPlatform()
        disconnect(PlatformSettings.Connection)
        PlatformSettings.Connection = nil
        local platform = PlatformSettings.Part
        PlatformSettings.Part = nil
        PlatformSettings.UpHeld = false
        PlatformSettings.DownHeld = false
        PlatformSettings.ForwardHeld = false
        if platform and platform.Parent then
                pcall(function() platform:Destroy() end)
        end
end

local function getPlatformCharacterData()
        local character = LocalPlayer.Character
        if not character or not character.Parent then return nil, nil, nil end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local root = getRoot(character)
        if not humanoid or not humanoid.Parent or not root or not root.Parent then return nil, nil, nil end
        return character, humanoid, root
end

local function getPlatformStartCFrame()
        local _, humanoid, root = getPlatformCharacterData()
        if not humanoid or not root then return nil end
        local platformHalfHeight = PlatformSettings.Size.Y * 0.5
        local offset = humanoid.HipHeight + root.Size.Y * 0.5 + platformHalfHeight
        return CFrame.new(root.Position - Vector3.new(0, offset, 0))
end

local function shouldCarryCharacter(platform, root, humanoid)
        if not platform or not platform.Parent or not root or not root.Parent or not humanoid or not humanoid.Parent then return false end
        local point = platform.CFrame:PointToObjectSpace(root.Position)
        local expectedY = humanoid.HipHeight + root.Size.Y * 0.5 + platform.Size.Y * 0.5
        return math.abs(point.X) <= platform.Size.X * 0.5 + 1 and math.abs(point.Z) <= platform.Size.Z * 0.5 + 1 and math.abs(point.Y - expectedY) <= 6
end

local function movePlatformBy(offset)
        local platform = PlatformSettings.Part
        if not platform or not platform.Parent then return end
        platform.CFrame = platform.CFrame + offset
        local _, humanoid, root = getPlatformCharacterData()
        if shouldCarryCharacter(platform, root, humanoid) then
                root.CFrame = root.CFrame + offset
                root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        end
end

local function getPlatformMoveDirection()
        local direction = Vector3.new(0, 0, 0)
        if PlatformSettings.UpHeld then direction += Vector3.new(0, 1, 0) end
        if PlatformSettings.DownHeld then direction -= Vector3.new(0, 1, 0) end
        if PlatformSettings.ForwardHeld then
                local camera = Workspace.CurrentCamera
                local look = camera and camera.CFrame.LookVector or Vector3.new(0, 0, -1)
                local forward = Vector3.new(look.X, 0, look.Z)
                if forward.Magnitude > 0.001 then direction += forward.Unit end
        end
        return direction
end

local function updatePlatform(deltaTime)
        if not running or not PlatformSettings.Enabled or not PlatformSettings.Part or not PlatformSettings.Part.Parent then return end
        local duration = math.max(tonumber(deltaTime) or 0, 0)
        if duration <= 0 then return end
        local direction = getPlatformMoveDirection()
        if direction.Magnitude > 0.001 then
                movePlatformBy(direction.Unit * PlatformSettings.MoveSpeed * duration)
        end
        local platform = PlatformSettings.Part
        local _, humanoid, root = getPlatformCharacterData()
        if not platform or not platform.Parent or not humanoid or not root then return end
        local footOffset = humanoid.HipHeight + root.Size.Y * 0.5 + platform.Size.Y * 0.5
        local targetPosition = root.Position - Vector3.new(0, footOffset, 0)
        local currentPosition = platform.Position
        local horizontalDelta = Vector3.new(targetPosition.X - currentPosition.X, 0, targetPosition.Z - currentPosition.Z)
        local horizontalDistance = horizontalDelta.Magnitude
        local verticalDelta = targetPosition.Y - currentPosition.Y
        if horizontalDistance > PlatformSettings.TrackSnapDistance or math.abs(verticalDelta) > PlatformSettings.TrackSnapDistance then
                platform.CFrame = CFrame.new(targetPosition)
                return
        end
        local horizontalOffset = Vector3.new(0, 0, 0)
        if horizontalDistance > 0.001 then
                local horizontalStep = math.min(horizontalDistance, PlatformSettings.TrackHorizontalSpeed * duration)
                horizontalOffset = horizontalDelta.Unit * horizontalStep
        end
        local downwardVelocity = math.max(-root.AssemblyLinearVelocity.Y, 0)
        local verticalSpeed = PlatformSettings.TrackUpSpeed
        if verticalDelta < 0 then
                verticalSpeed = PlatformSettings.TrackDownSpeed + downwardVelocity * 5
                if root.Position.Y < currentPosition.Y - 3 then
                        verticalSpeed = math.max(verticalSpeed, PlatformSettings.TrackSwoopSpeed + downwardVelocity * 6)
                end
        end
        local verticalOffset = math.clamp(verticalDelta, -verticalSpeed * duration, verticalSpeed * duration)
        if horizontalOffset.Magnitude > 0.001 or math.abs(verticalOffset) > 0.001 then
                platform.CFrame = platform.CFrame + horizontalOffset + Vector3.new(0, verticalOffset, 0)
        end
end

local function startPlatformForCharacter()
        clearPlatform()
        if not running or not PlatformSettings.Enabled then return end
        local cframe = getPlatformStartCFrame()
        if not cframe then return end
        local platform = Instance.new("Part")
        platform.Name = "__PlayerToolsPlatform"
        platform.Anchored = true
        platform.CanCollide = true
        platform.CanTouch = true
        platform.CanQuery = false
        platform.Size = PlatformSettings.Size
        platform.CFrame = cframe
        platform.Material = Enum.Material.Plastic
        platform.Color = Color3.fromRGB(160, 0, 255)
        platform.TopSurface = Enum.SurfaceType.Studs
        platform.Parent = Workspace
        PlatformSettings.Part = platform
        PlatformSettings.Connection = RunService.Heartbeat:Connect(function(deltaTime)
                local ok = pcall(updatePlatform, deltaTime)
                if not ok then clearPlatform() end
        end)
end

local function setPlatformEnabled(value)
        PlatformSettings.Enabled = value and true or false
        if not PlatformSettings.Enabled then
                clearPlatform()
                return
        end
        startPlatformForCharacter()
end

local function findVehicleModel(seat)
        local current = seat and seat.Parent
        while current and current ~= Workspace do
                if current:IsA("Model") then return current end
                current = current.Parent
        end
        return nil
end

local function getVehicleRoot(seat, model)
        if seat and seat:IsA("BasePart") then
                local assemblyRoot = seat.AssemblyRootPart
                if assemblyRoot and assemblyRoot:IsA("BasePart") then return assemblyRoot end
        end
        if model and model.PrimaryPart and model.PrimaryPart:IsA("BasePart") then return model.PrimaryPart end
        if model then
                for _, object in ipairs(model:GetDescendants()) do
                        if object:IsA("BasePart") then
                                local assemblyRoot = object.AssemblyRootPart
                                if assemblyRoot and assemblyRoot:IsA("BasePart") then return assemblyRoot end
                                return object
                        end
                end
        end
        return seat
end

local function isVehicleSeatPart(seat)
        return seat and seat:IsA("BasePart") and (seat:IsA("Seat") or seat:IsA("VehicleSeat"))
end

local function clearVehicleFlipAssist()
        if VehicleSettings.FlipGyro then
                pcall(function() VehicleSettings.FlipGyro:Destroy() end)
        end
        VehicleSettings.FlipGyro = nil
        VehicleSettings.FlipRoot = nil
end

local function getVehicleUprightCFrame(root, seat)
        local heading = Vector3.new(seat.CFrame.LookVector.X, 0, seat.CFrame.LookVector.Z)
        if heading.Magnitude < 0.001 then
                heading = Vector3.new(root.CFrame.LookVector.X, 0, root.CFrame.LookVector.Z)
        end
        if heading.Magnitude < 0.001 then
                heading = Vector3.zAxis
        else
                heading = heading.Unit
        end
        return CFrame.lookAt(root.Position, root.Position + heading, Vector3.yAxis)
end

local function restoreVehicleSpeed()
        clearVehicleSpeedWatch()
        clearVehicleBoost()
        clearVehicleFlipAssist()
        local seat = VehicleSettings.CurrentSeat
        if seat and seat.Parent and seat:IsA("VehicleSeat") then
                pcall(function()
                        if VehicleSettings.OriginalMaxSpeed[seat] ~= nil then
                                seat.MaxSpeed = VehicleSettings.OriginalMaxSpeed[seat]
                        end
                        if VehicleSettings.OriginalTurnSpeed[seat] ~= nil then
                                seat.TurnSpeed = VehicleSettings.OriginalTurnSpeed[seat]
                        end
                end)
        end
        VehicleSettings.CurrentSeat = nil
        VehicleSettings.CurrentModel = nil
        VehicleSettings.CurrentRoot = nil
        VehicleSettings.Updating = false
end

local function applyVehicleSeatSpeed()
        local seat = VehicleSettings.CurrentSeat
        if not running or not seat or not seat.Parent or not seat:IsA("VehicleSeat") then return end
        local speed = math.clamp(tonumber(VehicleSettings.Speed) or 60, 0, 500)
        pcall(function()
                if VehicleSettings.OriginalMaxSpeed[seat] == nil then
                        VehicleSettings.OriginalMaxSpeed[seat] = seat.MaxSpeed
                end
                if seat.MaxSpeed ~= speed then
                        VehicleSettings.Updating = true
                        seat.MaxSpeed = speed
                        VehicleSettings.Updating = false
                end
        end)
end

local function applyVehicleSeatSteering()
        local seat = VehicleSettings.CurrentSeat
        if not running or not seat or not seat.Parent or not seat:IsA("VehicleSeat") then return end
        local strength = math.clamp(tonumber(VehicleSettings.SteeringStrength) or 8, 0, 50)
        pcall(function()
                if VehicleSettings.OriginalTurnSpeed[seat] == nil then
                        VehicleSettings.OriginalTurnSpeed[seat] = seat.TurnSpeed
                end
                if seat.TurnSpeed ~= strength then
                        VehicleSettings.Updating = true
                        seat.TurnSpeed = strength
                        VehicleSettings.Updating = false
                end
        end)
end

local function getVehicleSteer(seat)
        if seat and seat:IsA("VehicleSeat") then
                local steer = 0
                pcall(function() steer = tonumber(seat.SteerFloat) or 0 end)
                if math.abs(steer) > 0.01 then return math.clamp(steer, -1, 1) end
                pcall(function() steer = tonumber(seat.Steer) or 0 end)
                if math.abs(steer) > 0.01 then return math.clamp(steer, -1, 1) end
        end
        local steer = 0
        if UserInputService:IsKeyDown(Enum.KeyCode.D) or UserInputService:IsKeyDown(Enum.KeyCode.Right) then steer += 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) or UserInputService:IsKeyDown(Enum.KeyCode.Left) then steer -= 1 end
        return math.clamp(steer, -1, 1)
end

local function stabilizeVehicle(root, seat, deltaTime)
        local upVector = root.CFrame.UpVector
        local needsCorrection = upVector.Y < 0.35
        local gyro = VehicleSettings.FlipGyro
        if not needsCorrection and gyro and upVector.Y >= 0.88 then
                clearVehicleFlipAssist()
                return
        end
        if not needsCorrection and not gyro then return end
        if not gyro or not gyro.Parent or VehicleSettings.FlipRoot ~= root then
                clearVehicleFlipAssist()
                gyro = Instance.new("BodyGyro")
                gyro.Name = "__PlayerToolsVehicleFlipStabilizer"
                gyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
                gyro.P = 50000
                gyro.D = 2200
                gyro.Parent = root
                VehicleSettings.FlipGyro = gyro
                VehicleSettings.FlipRoot = root
        end
        gyro.CFrame = getVehicleUprightCFrame(root, seat)
        local angularVelocity = root.AssemblyAngularVelocity
        root.AssemblyAngularVelocity = Vector3.new(angularVelocity.X * 0.2, angularVelocity.Y * 0.8, angularVelocity.Z * 0.2)
end

local function getVehicleThrottle(seat)
        if seat and seat:IsA("VehicleSeat") then
                local throttle = 0
                pcall(function() throttle = tonumber(seat.ThrottleFloat) or 0 end)
                if math.abs(throttle) > 0.01 then return math.clamp(throttle, -1, 1) end
                pcall(function() throttle = tonumber(seat.Throttle) or 0 end)
                if math.abs(throttle) > 0.01 then return math.clamp(throttle, -1, 1) end
        end
        local throttle = 0
        if UserInputService:IsKeyDown(Enum.KeyCode.W) or UserInputService:IsKeyDown(Enum.KeyCode.Up) then throttle += 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) or UserInputService:IsKeyDown(Enum.KeyCode.Down) then throttle -= 1 end
        return math.clamp(throttle, -1, 1)
end

local function updateVehicleBoost(deltaTime)
        if not running then return end
        local seat = VehicleSettings.CurrentSeat
        local root = VehicleSettings.CurrentRoot
        local humanoid = MovementSettings.Humanoid
        if not seat or not seat.Parent or not root or not root.Parent or not humanoid or not humanoid.Parent or not isVehicleSeatPart(seat) or seat.Occupant ~= humanoid then
                clearVehicleFlipAssist()
                return
        end
        local delta = math.max(tonumber(deltaTime) or 0, 0)
        applyVehicleSeatSpeed()
        applyVehicleSeatSteering()
        local forward = Vector3.new(seat.CFrame.LookVector.X, 0, seat.CFrame.LookVector.Z)
        if forward.Magnitude > 0.001 then
                forward = forward.Unit
                local throttle = getVehicleThrottle(seat)
                local speed = math.clamp(tonumber(VehicleSettings.Speed) or 60, 0, 500)
                local velocity = root.AssemblyLinearVelocity
                local forwardSpeed = velocity:Dot(forward)
                local targetSpeed = throttle * speed
                local alpha = 1 - math.exp(-18 * delta)
                local nextForwardSpeed = forwardSpeed + (targetSpeed - forwardSpeed) * alpha
                local lateralVelocity = velocity - forward * forwardSpeed
                root.AssemblyLinearVelocity = lateralVelocity + forward * nextForwardSpeed
                local steer = getVehicleSteer(seat)
                local horizontalSpeed = Vector3.new(velocity.X, 0, velocity.Z).Magnitude
                local movementFactor = math.clamp(horizontalSpeed / math.max(speed, 1), 0, 1)
                local targetYaw = -steer * math.clamp(tonumber(VehicleSettings.SteeringStrength) or 8, 0, 50) * movementFactor
                local angularVelocity = root.AssemblyAngularVelocity
                local steeringAlpha = 1 - math.exp(-14 * delta)
                root.AssemblyAngularVelocity = Vector3.new(angularVelocity.X, angularVelocity.Y + (targetYaw - angularVelocity.Y) * steeringAlpha, angularVelocity.Z)
        end
        stabilizeVehicle(root, seat, delta)
end

local function startVehicleBoost()
        clearVehicleBoost()
        VehicleSettings.BoostConnection = RunService.Heartbeat:Connect(updateVehicleBoost)
end

local function setVehicleSeat(seat)
        if seat == VehicleSettings.CurrentSeat then
                applyVehicleSeatSpeed()
                return
        end
        restoreVehicleSpeed()
        if not isVehicleSeatPart(seat) then return end
        VehicleSettings.CurrentSeat = seat
        VehicleSettings.CurrentModel = findVehicleModel(seat)
        VehicleSettings.CurrentRoot = getVehicleRoot(seat, VehicleSettings.CurrentModel)
        if not VehicleSettings.CurrentRoot or not VehicleSettings.CurrentRoot:IsA("BasePart") then
                restoreVehicleSpeed()
                return
        end
        if seat:IsA("VehicleSeat") then
                VehicleSettings.OriginalMaxSpeed[seat] = seat.MaxSpeed
                VehicleSettings.OriginalTurnSpeed[seat] = seat.TurnSpeed
                VehicleSettings.SpeedWatchConnection = seat:GetPropertyChangedSignal("MaxSpeed"):Connect(function()
                        if running and VehicleSettings.CurrentSeat == seat and not VehicleSettings.Updating and seat.MaxSpeed ~= VehicleSettings.Speed then
                                task.defer(applyVehicleSeatSpeed)
                        end
                end)
        end
        applyVehicleSeatSpeed()
        applyVehicleSeatSteering()
        startVehicleBoost()
end

local function bindVehicleHumanoid(humanoid)
        clearVehicleSeatedConnection()
        restoreVehicleSpeed()
        if not running or not humanoid or not humanoid.Parent then return end
        VehicleSettings.SeatedConnection = humanoid.Seated:Connect(function(isSeated, seatPart)
                if isSeated then setVehicleSeat(seatPart) else setVehicleSeat(nil) end
        end)
        setVehicleSeat(humanoid.SeatPart)
end

local function bindMovementCharacter(character)
        clearSpeedWatcher()
        clearJumpWatcher()
        MovementSettings.Humanoid = nil
        MovementSettings.OriginalSpeed = nil
        MovementSettings.OriginalJumpPower = nil
        if not character then return end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if not humanoid then humanoid = character:WaitForChild("Humanoid", 10) end
        if not running or not humanoid or not humanoid:IsA("Humanoid") then return end
        MovementSettings.Humanoid = humanoid
        bindVehicleHumanoid(humanoid)
        if MovementSettings.Noclip then applyNoclip() end
        if PlatformSettings.Enabled then
                task.defer(function()
                        if running and PlatformSettings.Enabled and LocalPlayer.Character == character then
                                startPlatformForCharacter()
                        end
                end)
        end
        if MovementSettings.Enabled then
                MovementSettings.OriginalSpeed = humanoid.WalkSpeed
                applySpeed()
                watchSpeed()
        end
        if MovementSettings.JumpEnabled then
                MovementSettings.OriginalJumpPower = humanoid.JumpPower
                applyJump()
                watchJump()
        end
        if AntiSitSettings.Enabled then       -- ⬅️ เพิ่ม 3 บรรทัดนี้
                enableAntiSit()
        end
end

local function ColorFromHSV(h, s, v)
    return Color3.fromHSV(h, s, v)
end

local function clearRGB()
    disconnect(RGBSettings.Connection)
    RGBSettings.Connection = nil
    
    -- คืนสีเดิม
    for part, originalColor in pairs(RGBSettings.OriginalColors) do
        if part and part.Parent then
            pcall(function() part.Color = originalColor end)
        end
    end
    RGBSettings.OriginalColors = setmetatable({}, {__mode = "k"})
    RGBSettings.Hue = 0
end

local function applyRGB()
    local char = LocalPlayer.Character
    if not char then return end
    
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and not part:IsA("Accessory") then
            -- เก็บสีเดิมครั้งแรก
            if RGBSettings.OriginalColors[part] == nil then
                RGBSettings.OriginalColors[part] = part.Color
            end
            
            -- เปลี่ยนเป็นสีรุ้ง
            local hue = (RGBSettings.Hue + (part.Position.Y * 0.05)) % 1
            pcall(function()
                part.Color = ColorFromHSV(hue, 1, 1)
            end)
        end
    end
end

local function startRGB()
    clearRGB()
    if not RGBSettings.Enabled then return end
    
    RGBSettings.Hue = 0
    applyRGB()
    
    RGBSettings.Connection = RunService.Heartbeat:Connect(function(dt)
        if not RGBSettings.Enabled then return end
        
        local speed = math.clamp(tonumber(RGBSettings.Speed) or 1, 0.1, 10)
        RGBSettings.Hue = (RGBSettings.Hue + dt * speed * 0.3) % 1
        
        applyRGB()
    end)
end

local function setRGBEnabled(value)
    RGBSettings.Enabled = value and true or false
    if RGBSettings.Enabled then
        startRGB()
    else
        clearRGB()
    end
end

local function clearAura()
    disconnect(AuraSettings.Connection)
    AuraSettings.Connection = nil
    
    if AuraSettings.Particle then
        pcall(function() AuraSettings.Particle:Destroy() end)
        AuraSettings.Particle = nil
    end
    
    if AuraSettings.Light then
        pcall(function() AuraSettings.Light:Destroy() end)
        AuraSettings.Light = nil
    end
end

local function createAuraOnCharacter()
    local char = LocalPlayer.Character
    if not char then return end
    
    local hrp = char:FindFirstChild("HumanoidRootPart") 
                or char:FindFirstChild("UpperTorso")
                or char:FindFirstChild("Torso")
    if not hrp then return end
    
    clearAura()

    local emitter = Instance.new("ParticleEmitter")
    emitter.Name = "__AuraParticle"
    emitter.Texture = "rbxassetid://243660364"  -- ดาวสว่าง
    emitter.LightEmission = 1
    emitter.LightInfluence = 0
    emitter.Color = ColorSequence.new(AuraSettings.Color)
    emitter.Size = NumberSequence.new({

        NumberSequenceKeypoint.new(0.3, AuraSettings.Size * 0.5),
        NumberSequenceKeypoint.new(1, 0)
    })
    emitter.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, AuraSettings.Transparency),
        NumberSequenceKeypoint.new(1, 1)
    })
    emitter.Lifetime = NumberRange.new(1, 2)
    emitter.Rate = AuraSettings.Rate

    emitter.SpreadAngle = Vector2.new(180, 180)

    light.Name = "__AuraLight"
    light.Color = AuraSettings.Color
    light.Brightness = 2
    light.Range = AuraSettings.Size * 2
    light.Parent = hrp
    AuraSettings.Light = light
end

local function startAura()
    clearAura()
    if not AuraSettings.Enabled then return end
    
    createAuraOnCharacter()
    
    -- ติดตามการ respawn
    AuraSettings.Connection = LocalPlayer.CharacterAdded:Connect(function()
        task.wait(0.5)
        if AuraSettings.Enabled then
            createAuraOnCharacter()
        end
    end)
end

local function setAuraEnabled(value)
    AuraSettings.Enabled = value and true or false
    if AuraSettings.Enabled then
        startAura()
    else
        clearAura()
    end
end

local function updateAura()
    if not AuraSettings.Enabled then return end
    if AuraSettings.Particle and AuraSettings.Particle.Parent then
        AuraSettings.Particle.Color = ColorSequence.new(AuraSettings.Color)
        AuraSettings.Particle.Size = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0),
            NumberSequenceKeypoint.new(0.3, AuraSettings.Size * 0.5),
            NumberSequenceKeypoint.new(1, 0)
        })
        AuraSettings.Particle.Rate = AuraSettings.Rate
        AuraSettings.Particle.Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, AuraSettings.Transparency),
            NumberSequenceKeypoint.new(1, 1)
        })
    end
    if AuraSettings.Light and AuraSettings.Light.Parent then
        AuraSettings.Light.Color = AuraSettings.Color
        AuraSettings.Light.Range = AuraSettings.Size * 2
    end
end

        local function getWallRaycastResult()
                local character = LP.Character
                if not character then return nil end
                local humanoidRootPart = character:FindFirstChild("HumanoidRootPart")
                if not humanoidRootPart then return nil end
                raycastParams.FilterDescendantsInstances = {character}
                local detectionDistance = 2

                local hrpCF = humanoidRootPart.CFrame
                for i = 0, 7 do
                        local angle = math.rad(i * 45)
                        local direction = (hrpCF * CFrame.Angles(0, angle, 0)).LookVector
                        local ray = Workspace:Raycast(humanoidRootPart.Position, direction * detectionDistance, raycastParams)
                        if ray and ray.Instance and ray.Distance < minDistance then
                                minDistance = ray.Distance
                                closestHit = ray
                        end
                end
                local blockCastSize = Vector3.new(1.5, 1, 0.5)
                local blockCastOffset = CFrame.new(0, -1, -0.5)
                local blockCastOriginCF = hrpCF * blockCastOffset
                local blockCastDirection = hrpCF.LookVector
                local blockCastDistance = 1.5
                local blockResult = Workspace:Blockcast(blockCastOriginCF, blockCastSize, blockCastDirection * blockCastDistance, raycastParams)
                if blockResult and blockResult.Instance and blockResult.Distance < minDistance then
                        minDistance = blockResult.Distance
                        closestHit = blockResult
                end
                return closestHit
        end

        local function executeWallJump(wallRayResult, jumpType)
                if jumpType ~= "Button" and not InfiniteJumpEnabled then return end
                local character = LP.Character
                local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                local rootPart = character and character:FindFirstChild("HumanoidRootPart")
                local camera = Workspace.CurrentCamera
                if not (humanoid and rootPart and camera and humanoid:GetState() ~= Enum.HumanoidStateType.Dead and wallRayResult) then return end
                if jumpType ~= "Button" then InfiniteJumpEnabled = false end
                local maxInfluenceAngleRight = math.rad(20)
                local maxInfluenceAngleLeft = math.rad(-100)
                local wallNormal = wallRayResult.Normal
                local baseDirectionAwayFromWall = Vector3.new(wallNormal.X, 0, wallNormal.Z).Unit
                if baseDirectionAwayFromWall.Magnitude < 0.1 then
                        local dirToHit = (wallRayResult.Position - rootPart.Position) * Vector3.new(1, 0, 1)
                        baseDirectionAwayFromWall = -dirToHit.Unit
                        if baseDirectionAwayFromWall.Magnitude < 0.1 then
                                baseDirectionAwayFromWall = -rootPart.CFrame.LookVector * Vector3.new(1, 0, 1)
                                if baseDirectionAwayFromWall.Magnitude > 0.1 then
                                        baseDirectionAwayFromWall = baseDirectionAwayFromWall.Unit
                                end
                                if baseDirectionAwayFromWall.Magnitude < 0.1 then
                                        baseDirectionAwayFromWall = Vector3.new(0, 0, 1)
                                end
                        end
                end
                baseDirectionAwayFromWall = Vector3.new(baseDirectionAwayFromWall.X, 0, baseDirectionAwayFromWall.Z).Unit
                if baseDirectionAwayFromWall.Magnitude < 0.1 then
                        baseDirectionAwayFromWall = Vector3.new(0, 0, 1)
                end
                local cameraLook = camera.CFrame.LookVector
                local horizontalCameraLook = Vector3.new(cameraLook.X, 0, cameraLook.Z).Unit
                if horizontalCameraLook.Magnitude < 0.1 then
                        horizontalCameraLook = baseDirectionAwayFromWall
                end
                local dot = math.clamp(baseDirectionAwayFromWall:Dot(horizontalCameraLook), -1, 1)
                local angleBetween = math.acos(dot)
                local cross = baseDirectionAwayFromWall:Cross(horizontalCameraLook)
                local rotationSign = -math.sign(cross.Y)
                if rotationSign == 0 then angleBetween = 0 end
                local actualInfluenceAngle
                if rotationSign == 1 then
                        actualInfluenceAngle = math.min(angleBetween, maxInfluenceAngleRight)
                elseif rotationSign == -1 then
                        actualInfluenceAngle = math.min(angleBetween, maxInfluenceAngleLeft)
                else
                        actualInfluenceAngle = 0
                end
                local adjustmentRotation = CFrame.Angles(0, actualInfluenceAngle * rotationSign, 0)
                local initialTargetLookDirection = adjustmentRotation * baseDirectionAwayFromWall
                rootPart.CFrame = CFrame.lookAt(rootPart.Position, rootPart.Position + initialTargetLookDirection)
                RunService.Heartbeat:Wait()
                local didJump = false
                if humanoid and humanoid:GetState() ~= Enum.HumanoidStateType.Dead then
                        humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
                        didJump = true
                        rootPart.CFrame = rootPart.CFrame * CFrame.Angles(0, -1, 0)
                        task.wait(0.15)
                        rootPart.CFrame = rootPart.CFrame * CFrame.Angles(0, 1, 0)
                end
                if didJump then
                        local directionTowardsWall = -baseDirectionAwayFromWall
                        task.wait(0.05)
                        rootPart.CFrame = CFrame.lookAt(rootPart.Position, rootPart.Position + directionTowardsWall)
                end
                if jumpType ~= "Button" then
                        task.wait(0.1)
                        InfiniteJumpEnabled = true
                end
        end

        local function SkidFling(TargetPlayer)
                local Character = LP.Character
                local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
                local RootPart = Humanoid and Humanoid.RootPart
                local TCharacter = TargetPlayer.Character
                if not TCharacter then return end

                local THumanoid
                local TRootPart
                local THead
                local Accessory
                local Handle
                if TCharacter:FindFirstChildOfClass("Humanoid") then THumanoid = TCharacter:FindFirstChildOfClass("Humanoid") end
                if THumanoid and THumanoid.RootPart then TRootPart = THumanoid.RootPart end
                if TCharacter:FindFirstChild("Head") then THead = TCharacter.Head end
                if TCharacter:FindFirstChildOfClass("Accessory") then Accessory = TCharacter:FindFirstChildOfClass("Accessory") end
                if Accessory and Accessory:FindFirstChild("Handle") then Handle = Accessory.Handle end

                if Character and Humanoid and RootPart then
                        if RootPart.Velocity.Magnitude < 50 then
                                getgenv().OldPos = RootPart.CFrame
                        end
                        if THumanoid and THumanoid.Sit then return end
                        if THead then
                                workspace.CurrentCamera.CameraSubject = THead
                        elseif Handle then
                                workspace.CurrentCamera.CameraSubject = Handle
                        elseif THumanoid and TRootPart then
                                workspace.CurrentCamera.CameraSubject = THumanoid
                        end
                        if not TCharacter:FindFirstChildWhichIsA("BasePart") then return end

                        local FPos = function(BasePart, Pos, Ang)
                                RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
                                Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
                                RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
                                RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
                        end

                        local SFBasePart = function(BasePart)
                                local TimeToWait = 2
                                local Time = tick()
                                local Angle = 0
                                repeat
                                if RootPart and THumanoid then
                                        if BasePart.Velocity.Magnitude < 50 then
                                                Angle = Angle + 100
                                                FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                                                task.wait()
                                                FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                                                task.wait()
                                                FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                                                task.wait()
                                                FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                                                task.wait()
                                                FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0))
                                                task.wait()
                                                FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0))
                                                task.wait()
                                        else
                                                FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                                                task.wait()
                                                FPos(BasePart, CFrame.new(0, -1.5, -THumanoid.WalkSpeed), CFrame.Angles(0, 0, 0))
                                                task.wait()
                                                FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                                                task.wait()
                                                FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
                                                task.wait()
                                                FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                                                task.wait()
                                                FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
                                                task.wait()
                                                FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                                                task.wait()
                                        end
                                end
                                until Time + TimeToWait < tick() or not FlingActive
                        end

                        workspace.FallenPartsDestroyHeight = 0/0

                        local BV = Instance.new("BodyVelocity")
                        BV.Parent = RootPart
                        BV.Velocity = Vector3.new(0, 0, 0)
                        BV.MaxForce = Vector3.new(9e9, 9e9, 9e9)

                        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

                        if TRootPart then
                                SFBasePart(TRootPart)
                        elseif THead then
                                SFBasePart(THead)
                        elseif Handle then
                                SFBasePart(Handle)
                        else
                                return
                        end

                        BV:Destroy()
                        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
                        workspace.CurrentCamera.CameraSubject = Humanoid

                        if getgenv().OldPos then
                                repeat
                                        RootPart.CFrame = getgenv().OldPos * CFrame.new(0, .5, 0)
                                        Character:SetPrimaryPartCFrame(getgenv().OldPos * CFrame.new(0, .5, 0))
                                        Humanoid:ChangeState("GettingUp")
                                        for _, part in pairs(Character:GetChildren()) do
                                                if part:IsA("BasePart") then
                                                        part.Velocity, part.RotVelocity = Vector3.new(), Vector3.new()
                                                end
                                        end
                                        task.wait()
                                until (RootPart.Position - getgenv().OldPos.p).Magnitude < 25
                                workspace.FallenPartsDestroyHeight = getgenv().FPDH
                        end
                end
        end

        local function StartFling()
                if FlingActive then return end
                local count = 0
                for _ in pairs(SelectedTargets) do count = count + 1 end
                if count == 0 then
                        statusLabel.Text = "ยังไม่ได้เลือกเป้าหมาย!"
                        task.wait(1)
                        UpdateStatusText()
                        return
                end
                FlingActive = true
                UpdateStatusText()
                task.spawn(function()
                        while FlingActive do
                                local validTargets = {}
                                for name, player in pairs(SelectedTargets) do
                                        if player and player.Parent then
                                                validTargets[name] = player
                                        else
                                                SelectedTargets[name] = nil
                                                local cb = PlayerCheckboxes[name]
                                                if cb then cb.Checkmark.Visible = false end
                                        end
                                end
                                for _, player in pairs(validTargets) do
                                        if FlingActive then
                                                pcall(SkidFling, player)
                                                task.wait(0.1)
                                        else
                                                break
                                        end
                                end
                                UpdateStatusText()
                                task.wait(0.5)
                        end
                end)
        end

        local function StopFling()
                if not FlingActive then return end
                FlingActive = false
                UpdateStatusText()
        end

        startButton.MouseButton1Click:Connect(StartFling)
        stopButton.MouseButton1Click:Connect(StopFling)
        selectAllButton.MouseButton1Click:Connect(function() ToggleAllPlayers(true) end)
        deselectAllButton.MouseButton1Click:Connect(function() ToggleAllPlayers(false) end)
        closeBtn.MouseButton1Click:Connect(function()
                StopFling()
                screenGui.Enabled = false
        end)

        Players.PlayerAdded:Connect(RefreshPlayerList)
        Players.PlayerRemoving:Connect(function(player)
                if SelectedTargets[player.Name] then
                        SelectedTargets[player.Name] = nil
                end
                RefreshPlayerList()
                UpdateStatusText()
        end)

        RefreshPlayerList()
        UpdateStatusText()

        screenGui.Destroying:Connect(function()
                FlingActive = false
                if keyConn then keyConn:Disconnect() end
        end)

        print("ฟลิปเป้าหมาย - โหลดแล้ว")
        SunlightFlingSettings.Loaded = true
end

local function getRandomPlasticColor()
    return PLASTIC_COLORS[math.random(1, #PLASTIC_COLORS)]
end

local function applyPlasticToPart(part)
    if not part or not part:IsA("BasePart") then return end
    if PlasticModeSettings.OriginalMaterials[part] == nil then
        PlasticModeSettings.OriginalMaterials[part] = part.Material
        PlasticModeSettings.OriginalColors[part] = part.Color
    end
    pcall(function()
        part.Material = Enum.Material.Plastic
        part.Color = getRandomPlasticColor()
        part.Reflectance = 0
        part.Transparency = 0
    end)
end

local function plasticProcessDescendant(obj)
    if obj:IsA("BasePart") and not obj:IsDescendantOf(LocalPlayer.Character or game) then
        applyPlasticToPart(obj)
    end
end

local function setPlasticModeEnabled(value)
    PlasticModeSettings.Enabled = value and true or false

    if PlasticModeSettings.Enabled then
        -- ✅ ใส่ Material Plastic ให้ทุก Part ที่มีอยู่
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("BasePart") then
                local char = LocalPlayer.Character
                if not (char and obj:IsDescendantOf(char)) then
                    applyPlasticToPart(obj)
                end
            end
        end

        -- ✅ เฝ้าดู Part ใหม่ที่ถูกสร้าง
        if PlasticModeSettings.Connection then
            pcall(function() PlasticModeSettings.Connection:Disconnect() end)
        end
        PlasticModeSettings.Connection = Workspace.DescendantAdded:Connect(function(obj)
            if not PlasticModeSettings.Enabled then return end
            if obj:IsA("BasePart") then
                task.defer(function()
                    local char = LocalPlayer.Character
                    if not (char and obj:IsDescendantOf(char)) then
                        applyPlasticToPart(obj)
                    end
                end)
            end
        end)

    else
        -- ✅ คืนค่าเดิม
        for part, originalMaterial in pairs(PlasticModeSettings.OriginalMaterials) do
            local originalColor = PlasticModeSettings.OriginalColors[part]
            if part and part.Parent then
                pcall(function()
                    part.Material = originalMaterial
                    if originalColor then
                        part.Color = originalColor
                    end
                end)
            end
        end
        PlasticModeSettings.OriginalMaterials = setmetatable({}, {__mode = "k"})
        PlasticModeSettings.OriginalColors = setmetatable({}, {__mode = "k"})

        if PlasticModeSettings.Connection then
            pcall(function() PlasticModeSettings.Connection:Disconnect() end)
            PlasticModeSettings.Connection = nil
        end
    end
end

        FPSBoostSettings.Enabled = value and true or false

-- ===== SPIN CHARACTER (BodyVelocity — ไม่ล้ม + เดินเร็วปกติ) =====
-- =========================================================

local function clearSpin()
    -- Disconnect loop
    disconnect(SpinSettings.Connection)
    SpinSettings.Connection = nil
    
    -- ลบ BodyVelocity
    if SpinSettings.BV then
        pcall(function() SpinSettings.BV:Destroy() end)
        SpinSettings.BV = nil
    end
    
    -- คืน state
    local char = LocalPlayer.Character
    local humanoid = char and char:FindFirstChildOfClass("Humanoid")
    if humanoid then
        pcall(function()
            if SpinStates.SavedStates then
                for state, enabled in pairs(SpinStates.SavedStates) do
                    humanoid:SetStateEnabled(state, enabled)
                end
            end
        end)
    end
    
    SpinStates.SavedStates = nil
    SpinStates.SpinAngle = 0
    SpinSettings.Root = nil
end

local function startSpin()
    clearSpin()
    if not running or not SpinSettings.Enabled then return end
    
    local char = LocalPlayer.Character
    if not char then return end
    
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not humanoid or not root then return end
    
    SpinSettings.Root = root
    
    -- ปิด State ที่ทำให้ล้ม
    SpinStates.SavedStates = {
        [Enum.HumanoidStateType.FallingDown] = humanoid:GetStateEnabled(Enum.HumanoidStateType.FallingDown),
        [Enum.HumanoidStateType.Ragdoll] = humanoid:GetStateEnabled(Enum.HumanoidStateType.Ragdoll),
        [Enum.HumanoidStateType.Physics] = humanoid:GetStateEnabled(Enum.HumanoidStateType.Physics),
        [Enum.HumanoidStateType.PlatformStanding] = humanoid:GetStateEnabled(Enum.HumanoidStateType.PlatformStanding),
    }
    
    pcall(function()
        humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
        humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
        humanoid:SetStateEnabled(Enum.HumanoidStateType.Physics, false)
        humanoid:SetStateEnabled(Enum.HumanoidStateType.PlatformStanding, false)
    end)
    
    SpinStates.SpinAngle = 0
    
    -- ✅ BodyVelocity ขับเคลื่อนแนวราบ (Bypass Humanoid speed limit)
    local bv = Instance.new("BodyVelocity")
    bv.Name = "__SpinBV"
    bv.MaxForce = Vector3.new(1e9, 0, 1e9)  -- เฉพาะแนวราบ
    bv.Velocity = Vector3.zero
    bv.Parent = root
    SpinSettings.BV = bv
    
    SpinSettings.Connection = RunService.RenderStepped:Connect(function(dt)
        if not SpinSettings.Enabled then return end
        
        local char = LocalPlayer.Character
        if not char then return end
        
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if not humanoid or not root then return end
        
        -- กันล้ม
        local state = humanoid:GetState()
        if state == Enum.HumanoidStateType.FallingDown 
            or state == Enum.HumanoidStateType.Ragdoll 
            or state == Enum.HumanoidStateType.Physics 
            or state == Enum.HumanoidStateType.PlatformStanding then
            pcall(function()
                humanoid:ChangeState(Enum.HumanoidStateType.Running)
            end)
        end
        
        -- ✅ ขับเคลื่อนด้วย BodyVelocity (ไม่ผ่าน Humanoid → ไม่ถูกจำกัดความเร็ว)
        local moveDir = humanoid.MoveDirection
        if moveDir.Magnitude > 0.01 and bv and bv.Parent then
            local speed = MovementSettings.Enabled and MovementSettings.Speed or 16
            bv.Velocity = moveDir * speed
        elseif bv and bv.Parent then
            bv.Velocity = Vector3.zero
        end
        
        -- หมุน
        local spinSpeed = math.clamp(tonumber(SpinSettings.Speed) or 10, 1, 500)
        SpinStates.SpinAngle = (SpinStates.SpinAngle + math.rad(spinSpeed * 60) * dt) % (math.pi * 2)
        
        local pos = root.Position
        root.CFrame = CFrame.new(pos) * CFrame.Angles(0, SpinStates.SpinAngle, 0)
    end)
end

local function setSpinEnabled(value)
    SpinSettings.Enabled = value and true or false
    if SpinSettings.Enabled then
        startSpin()
    else
        clearSpin()
    end
end

local function teleportToPlayer(playerName)
        local target = Players:FindFirstChild(playerName)
        if not target then return end
        local targetChar = target.Character
        local targetRoot = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
        local myChar = LocalPlayer.Character
        local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if not targetRoot or not myRoot then return end
        local now = os.clock()
        if now - TeleportPlayerSettings.LastTeleport < TeleportPlayerSettings.Cooldown then return end
        TeleportPlayerSettings.LastTeleport = now
        myRoot.CFrame = CFrame.new(targetRoot.Position + Vector3.new(0, 3, 0))
end

local function teleportToCursor()
        if not TeleportCursorSettings.Enabled then return end
        local now = os.clock()
        if now - TeleportCursorSettings.LastTeleport < TeleportCursorSettings.Cooldown then return end
        TeleportCursorSettings.LastTeleport = now
        local camera = Workspace.CurrentCamera
        local char = LocalPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not camera or not root then return end
        local mouseLocation = UserInputService:GetMouseLocation()
        local ray = camera:ScreenPointToRay(mouseLocation.X, mouseLocation.Y)
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = {char}
        local result = Workspace:Raycast(ray.Origin, ray.Direction * 10000, params)

local function setAntiFlingEnabled(value)
        AntiFlingSettings.Enabled = value and true or false
        if AntiFlingSettings.Enabled then
                local function watchPlayer(player)
                        if player == LocalPlayer then return end
                        local function lockCharacter(character)
                                for _, obj in ipairs(character:GetDescendants()) do
                                        if obj:IsA("BasePart") then
                                                if AntiFlingSettings.CollisionStates[obj] == nil then
                                                        AntiFlingSettings.CollisionStates[obj] = obj.CanCollide
                                                end
                                                pcall(function() obj.CanCollide = false end)
                                        end
                                end
                        end
                        if player.Character then lockCharacter(player.Character) end
                        table.insert(AntiFlingSettings.Connections, player.CharacterAdded:Connect(function(c)
                                if AntiFlingSettings.Enabled then lockCharacter(c) end
                        end))
                end
                for _, player in ipairs(Players:GetPlayers()) do watchPlayer(player) end
                table.insert(AntiFlingSettings.Connections, Players.PlayerAdded:Connect(watchPlayer))
        else
                for _, conn in ipairs(AntiFlingSettings.Connections) do disconnect(conn) end
                AntiFlingSettings.Connections = {}
                for part, orig in pairs(AntiFlingSettings.CollisionStates) do
                        if part and part.Parent then
                                pcall(function() part.CanCollide = orig end)
                        end
                end
                AntiFlingSettings.CollisionStates = setmetatable({}, {__mode = "k"})
        end
end

local function disableAntiSit()
    if AntiSitSettings.SitConnection then
        AntiSitSettings.SitConnection:Disconnect()
        AntiSitSettings.SitConnection = nil
    end
    if AntiSitSettings.StateConnection then
        AntiSitSettings.StateConnection:Disconnect()
        AntiSitSettings.StateConnection = nil
    end
    local humanoid = MovementSettings.Humanoid
    if humanoid and humanoid.Parent then
        pcall(function()
            humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
            humanoid.Sit = false
            if humanoid:GetState() == Enum.HumanoidStateType.Seated then
                humanoid:ChangeState(Enum.HumanoidStateType.Running)
            end
        end)
    end
end

local function enableAntiSit()
    disableAntiSit()
    local humanoid = MovementSettings.Humanoid
    if not humanoid or not humanoid.Parent then return end

    pcall(function()
        humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
        humanoid.Sit = false
    end)

    AntiSitSettings.StateConnection = humanoid.StateChanged:Connect(function(_, newState)
        if not AntiSitSettings.Enabled then return end
        if not humanoid or not humanoid.Parent then return end
        if newState == Enum.HumanoidStateType.Seated then
            pcall(function()
                humanoid.Sit = false
                humanoid:ChangeState(Enum.HumanoidStateType.Running)
            end)
        end
    end)

    AntiSitSettings.SitConnection = RunService.Stepped:Connect(function()
        if not AntiSitSettings.Enabled then return end
        if humanoid and humanoid.Parent and humanoid.Sit then
            pcall(function() humanoid.Sit = false end)
        end
    end)
end

-- บันทึก Config
local function saveConfig(name)
    name = name or "default"
    
    if not hasFileSupport() then
        warn("[Sunlight Hub] Executor ไม่รองรับการบันทึกไฟล์")
        return false, "ไม่รองรับการบันทึก"
    end
    
    if not ensureConfigFolder() then
        return false, "สร้าง folder ไม่ได้"
    end
    
    local data = collectConfig()
    
    local ok, encoded = pcall(function()
        return game:GetService("HttpService"):JSONEncode(data)
    end)
    
    if not ok or not encoded then
        warn("[Sunlight Hub] JSON Encode ล้มเหลว")
        return false, "JSON Encode ล้มเหลว"
    end
    
    local path = ConfigSettings.Folder .. "/" .. name .. ".json"
    local writeOk = pcall(function()
        writefile(path, encoded)
    end)
    
    if not writeOk then
        warn("[Sunlight Hub] เขียนไฟล์ล้มเหลว")
        return false, "เขียนไฟล์ล้มเหลว"
    end
    
    print("[Sunlight Hub] บันทึก config สำเร็จ:", name)
    return true, "บันทึกสำเร็จ"
end

-- โหลด Config
local function loadConfig(name)
    name = name or "default"
    
    if not hasFileSupport() then
        return false, "ไม่รองรับการอ่านไฟล์"
    end
    
    local path = ConfigSettings.Folder .. "/" .. name .. ".json"
    
    if not isfile(path) then
        return false, "ไม่พบไฟล์ config"
    end
    
    local ok, content = pcall(function()
        return readfile(path)
    end)
    
    if not ok or not content then
        return false, "อ่านไฟล์ล้มเหลว"
    end
    
    local decodeOk, data = pcall(function()
        return game:GetService("HttpService"):JSONDecode(content)
    end)
    
    if not decodeOk or type(data) ~= "table" then
        return false, "JSON Decode ล้มเหลว"
    end
    
    -- Apply configs (ด้วย pcall ทุกส่วน)
    pcall(function()
        -- WalkSpeed
        if data.WalkSpeed then
            MovementSettings.Speed = data.WalkSpeed.Speed or 16
            setSpeedEnabled(data.WalkSpeed.Enabled and true or false)
        end
        
        -- Jump
        if data.Jump then
            MovementSettings.JumpPower = data.Jump.JumpPower or 50
            MovementSettings.InfiniteJump = data.Jump.InfiniteJump and true or false
            setJumpEnabled(data.Jump.Enabled and true or false)
        end
        
        -- Noclip
        if data.Noclip ~= nil then
            setNoclipEnabled(data.Noclip and true or false)
        end
        
        -- Visuals
        if data.Fullbright ~= nil then setFullbrightEnabled(data.Fullbright) end
        if data.NoFog ~= nil then setNoFogEnabled(data.NoFog) end
        if data.FPSBoost ~= nil then setFPSBoostEnabled(data.FPSBoost) end
        if data.PlasticMode ~= nil then setPlasticModeEnabled(data.PlasticMode) end
        
        -- Time Changer
        if data.TimeChanger then
            TimeChangerSettings.ClockTime = data.TimeChanger.ClockTime or 14
            setTimeChangerEnabled(data.TimeChanger.Enabled and true or false)
        end
        
        -- Aimbot
        if data.Aimbot then
            AimbotSettings.Enabled = data.Aimbot.Enabled and true or false
            AimbotSettings.FOVPercent = data.Aimbot.FOVPercent or 20
            AimbotSettings.TeamCheck = data.Aimbot.TeamCheck and true or false
            AimbotSettings.AimPart = data.Aimbot.AimPart or "Head"
        end
        
        -- Hitbox
        if data.Hitbox then
            HitboxSettings.Size = data.Hitbox.Size or 12
            HitboxSettings.Transparency = data.Hitbox.Transparency or 0.6
            setHitboxEnabled(data.Hitbox.Enabled and true or false)
        end
        
        -- Utility
        if data.AntiAFK ~= nil then setAntiAFKEnabled(data.AntiAFK) end
        if data.AntiSit ~= nil then setAntiSitEnabled(data.AntiSit) end
        if data.AntiFling ~= nil then setAntiFlingEnabled(data.AntiFling) end
        
        if data.AutoClick then
            AutoClickSettings.Delay = data.AutoClick.Delay or 0.1
            setAutoClickEnabled(data.AutoClick.Enabled and true or false)
        end
        
        -- Gravity
        if data.Gravity then
            GravitySettings.Value = data.Gravity.Value or 196.2
            GravitySettings.Preset = data.Gravity.Preset or "Normal"
            setGravityEnabled(data.Gravity.Enabled and true or false)
        end
        
        -- Vehicle
        if data.Vehicle then
            VehicleSettings.Speed = data.Vehicle.Speed or 60
            VehicleSettings.SteeringStrength = data.Vehicle.SteeringStrength or 8
            applyVehicleSeatSpeed()
            applyVehicleSeatSteering()
        end
    end)
    
    print("[Sunlight Hub] โหลด config สำเร็จ:", name)
    return true, "โหลดสำเร็จ"
end

-- 2. ฟอร์แมตเวลา (เช่น "2d 3h 45m 12s")
local function formatServerAge(seconds)
    local d = math.floor(seconds / 86400)
    local h = math.floor((seconds % 86400) / 3600)
    local m = math.floor((seconds % 3600) / 60)
    local s = math.floor(seconds % 60)
    local str = ""
    if d > 0 then str = str .. d .. "d " end
    if h > 0 or d > 0 then str = str .. h .. "h " end
    str = str .. string.format("%02dm %02ds", m, s)
    return str
end

-- =========================================================
local function applyGodMode(character)
    if not character or not character.Parent then return end
    
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    
    -- เก็บค่าเดิม (ครั้งแรก)
    if not GodModeSettings.OriginalMaxHealth then
        GodModeSettings.OriginalMaxHealth = humanoid.MaxHealth
        GodModeSettings.OriginalHealth = humanoid.Health
    end
    
    -- ตั้งค่าเลือด
    pcall(function()
        humanoid.MaxHealth = GodModeSettings.MaxHealth
        humanoid.Health = GodModeSettings.MaxHealth
        humanoid.BreakJointsOnDeath = false
        humanoid.RequiresNeck = false
    end)
    
    -- ปิด Ragdoll
    if GodModeSettings.DisableRagdoll then
        pcall(function()
            humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
            humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
            humanoid:SetStateEnabled(Enum.HumanoidStateType.Physics, false)
            humanoid:SetStateEnabled(Enum.HumanoidStateType.PlatformStanding, false)
        end)
    end
    
    -- ForceField กันตาย
    if not character:FindFirstChild("__GodModeFF") then
        local ff = Instance.new("ForceField")
        ff.Name = "__GodModeFF"
        ff.Visible = false
        ff.Parent = character
    end
    
    -- Health Regeneration
    local healthConn = humanoid.HealthChanged:Connect(function(health)
        if not GodModeSettings.Enabled then return end
        if health < humanoid.MaxHealth then
            pcall(function()
                humanoid.Health = humanoid.MaxHealth
            end)
        end
    end)
    table.insert(GodModeSettings.Connections, healthConn)
    
    -- ตรวจจับ Kill Brick
    if GodModeSettings.AntiKillBrick then
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then
                local touchConn = part.Touched:Connect(function(hit)
                    if not GodModeSettings.Enabled then return end
                    if hit and hit:IsDescendantOf(character) then
                        -- ไม่ให้โดน Kill Brick
                        local hum = character:FindFirstChildOfClass("Humanoid")
                        if hum and hum.Health < hum.MaxHealth then
                            pcall(function()
                                hum.Health = hum.MaxHealth
                            end)
                        end
                    end
                end)
                table.insert(GodModeSettings.Connections, touchConn)
            end
        end
    end
end

local function setGodModeEnabled(value)
    GodModeSettings.Enabled = value and true or false
    
    -- Disconnect เก่า
    for _, conn in ipairs(GodModeSettings.Connections) do
        pcall(function() conn:Disconnect() end)
    end
    GodModeSettings.Connections = {}
    
    local char = LocalPlayer.Character
    
    if GodModeSettings.Enabled then
        -- เปิด
        if char then
            applyGodMode(char)
        end
        
        -- ผูกกับตัวละครใหม่
        local conn = LocalPlayer.CharacterAdded:Connect(function(newChar)
            if not GodModeSettings.Enabled then return end
            task.wait(0.5)
            applyGodMode(newChar)
        end)
        table.insert(GodModeSettings.Connections, conn)
        
        -- แจ้งเตือน
        pcall(function()
            game:GetService("StarterGui"):SetCore("SendNotification", {
                Title = "God Mode",
                Text = "Open God mode",
                Duration = 2
            })
        end)
    else
        -- ปิด - คืนค่าเดิม
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum and GodModeSettings.OriginalMaxHealth then
                pcall(function()
                    hum.MaxHealth = GodModeSettings.OriginalMaxHealth
                    hum.Health = math.min(hum.Health, GodModeSettings.OriginalMaxHealth)
                    hum.BreakJointsOnDeath = true
                    hum.RequiresNeck = true
                end)
            end
            
            -- คืน Ragdoll State
            if hum then
                pcall(function()
                    hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
                    hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true)
                    hum:SetStateEnabled(Enum.HumanoidStateType.Physics, true)
                    hum:SetStateEnabled(Enum.HumanoidStateType.PlatformStanding, true)
                end)
            end
            
            -- ลบ ForceField
            local ff = char:FindFirstChild("__GodModeFF")
            if ff then ff:Destroy() end
        end
        
        GodModeSettings.OriginalMaxHealth = nil
        GodModeSettings.OriginalHealth = nil
        
        pcall(function()
            game:GetService("StarterGui"):SetCore("SendNotification", {
                Title = " God Mode",
                Text = "Close God mode",
                Duration = 2
            })
        end)
    end
end

local function isNPC(model)
    if not model:IsA("Model") then return false end
    if Players:GetPlayerFromCharacter(model) then return false end
    local hum = model:FindFirstChildOfClass("Humanoid")
    if not hum then return false end
    return hum.Health > 0 and (model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Torso"))
end

local function createNPCESP(npc)
    if NPCESPSettings.Cache[npc] then return end

    local root = npc:FindFirstChild("HumanoidRootPart") or npc:FindFirstChild("Torso")
    if not root then return end

    -- Highlight
    local hl = Instance.new("Highlight")
    hl.Name = "__NPCESP_Highlight"
    hl.FillColor = NPCESPSettings.Color
    hl.FillTransparency = 0.7
    hl.OutlineColor = Color3.new(1, 1, 1)
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Adornee = npc
    hl.Parent = npc

    -- BillboardGui

    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true

    label.Size = UDim2.fromScale(1, 1)

        Label = label,
        Root = root,
        Hum = npc:FindFirstChildOfClass("Humanoid"),
        Billboard = billboard,
    }
end

local function removeNPCESP(npc)
    local data = NPCESPSettings.Cache[npc]
    if not data then return end

    local hl = npc:FindFirstChild("__NPCESP_Highlight")
    local bb = npc:FindFirstChild("__NPCESP_Label")
    if hl then hl:Destroy() end
    if bb then bb:Destroy() end

    NPCESPSettings.Cache[npc] = nil
end

local function clearAllNPCESP()
    for npc, _ in pairs(NPCESPSettings.Cache) do
        pcall(function()
            local hl = npc:FindFirstChild("__NPCESP_Highlight")
            local bb = npc:FindFirstChild("__NPCESP_Label")
            if hl then hl:Destroy() end
            if bb then bb:Destroy() end
        end)
    end
    NPCESPSettings.Cache = {}
end

local function scanAllNPCs()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if isNPC(obj) then
            createNPCESP(obj)
        end
    end
end

local function updateNPCESP()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")

    for npc, data in pairs(NPCESPSettings.Cache) do
        if not npc:IsDescendantOf(Workspace) or not data.Hum or data.Hum.Health <= 0 then
            removeNPCESP(npc)
        elseif hrp then
            local dist = (hrp.Position - data.Root.Position).Magnitude
            if NPCESPSettings.ShowDistance then
                data.Label.Text = string.format("[%d m]", math.floor(dist))
            else
                data.Label.Text = npc.Name
            end
        end
    end
end

local function setNPCESPEnabled(value)
    NPCESPSettings.Enabled = value and true or false

    -- Disconnect เก่า
    disconnect(NPCESPSettings.ScanConnection)
    disconnect(NPCESPSettings.UpdateConnection)
    NPCESPSettings.ScanConnection = nil
    NPCESPSettings.UpdateConnection = nil

    if not NPCESPSettings.Enabled then
        clearAllNPCESP()
        return
    end

    -- Scan รอบแรก
    scanAllNPCs()

    -- ติดตาม NPC ใหม่
    NPCESPSettings.ScanConnection = Workspace.DescendantAdded:Connect(function(obj)
        if not NPCESPSettings.Enabled then return end
        task.wait(0.1)
        if obj.Parent and isNPC(obj) then
            createNPCESP(obj)
        end
    end)

    -- Update ระยะทาง
    NPCESPSettings.UpdateConnection = RunService.Heartbeat:Connect(function()
        if not NPCESPSettings.Enabled then return end
        task.wait(NPCESPSettings.UpdateFrequency)
        updateNPCESP()
    end)
end

local PullSettings = {
    isBringing = false,
    connections = {}
}

local function getPullRootPart(character)
    return character:FindFirstChild("HumanoidRootPart")
        or character:FindFirstChild("Torso")
        or character:FindFirstChild("UpperTorso")
end

local function setPullCharacterCollision(character, state)
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CanCollide = state
        end
    end
end

local function isOnVehicleOrSeat(character)
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end
    if humanoid.SeatPart ~= nil then return true end
    for _, part in ipairs(character:GetChildren()) do
        if part:IsA("BasePart") then
            local ok, touching = pcall(function() return part:GetTouchingParts() end)
            if ok then
                for _, t in ipairs(touching) do
                    if t:IsA("VehicleSeat") or t:IsA("Seat") then return true end
                end
            end
        end
    end
    return false
end

local function unseatPlayer(character)
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        humanoid.Jump = true
        pcall(function() humanoid.SeatPart = nil end)
    end
end

local function startPullPlayers()
    if PullSettings.isBringing then return end
    PullSettings.isBringing = true

    for _, c in pairs(PullSettings.connections) do c:Disconnect() end
    PullSettings.connections = {}

    local loopConn = RunService.Heartbeat:Connect(function()
        if not PullSettings.isBringing then return end

        local myChar = LocalPlayer.Character
        if not myChar then return end
        local myRoot = getPullRootPart(myChar)
        if not myRoot then return end

        local myLockedCFrame = myRoot.CFrame
        local frontPosition = (myLockedCFrame * CFrame.new(0, 0, -3)).Position
        local targetCFrame = CFrame.new(frontPosition.X, myLockedCFrame.Position.Y, frontPosition.Z)

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                local otherChar = player.Character
                if otherChar then
                    local otherRoot = getPullRootPart(otherChar)
                    if otherRoot then
                        if isOnVehicleOrSeat(otherChar) then unseatPlayer(otherChar) end
                        setPullCharacterCollision(otherChar, false)
                        otherRoot.CFrame = targetCFrame
                    end
                end
            end
        end

        myRoot.CFrame = myLockedCFrame
    end)

    table.insert(PullSettings.connections, loopConn)
end

local function stopPullPlayers()
    PullSettings.isBringing = false
    for _, c in pairs(PullSettings.connections) do c:Disconnect() end
    PullSettings.connections = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local otherChar = player.Character
            if otherChar then setPullCharacterCollision(otherChar, true) end
        end
    end
end

do
    -- ✅ เพิ่มบรรทัดนี้
    local Camera = workspace.CurrentCamera

    -- FOV Circle (Drawing)
    local Circle = Drawing.new("Circle")
    Circle.Thickness = 1.5
    Circle.NumSides = 45
    Circle.Filled = false
    Circle.Transparency = 0.8
    Circle.Color = Color3.fromRGB(160, 0, 255)
    Circle.Visible = false
    AimbotSettings.Circle = Circle

    local function updateCircle()
        local vp = Camera.ViewportSize
        if vp.X == 0 or vp.Y == 0 then return end
        local baseRes = math.min(vp.X, vp.Y)
        Circle.Radius = (baseRes / 2) * (AimbotSettings.FOVPercent / 100)
        Circle.Position = Vector2.new(vp.X / 2, vp.Y / 2)
    end

    -- Team Check
    local function getPlayerTeam(player)
        if not player then return nil end
        if player.Team then return "Team:" .. player.Team.Name end
        local ok, color = pcall(function() return player.TeamColor end)
        if ok and color then return "Color:" .. tostring(color) end
        for _, name in ipairs({"Team", "TeamId", "TeamName", "Faction"}) do
            local v = player:GetAttribute(name)
            if v ~= nil then return name .. ":" .. tostring(v) end
        end
        local ls = player:FindFirstChild("leaderstats")
        if ls then
            for _, name in ipairs({"Team", "TeamId", "Faction"}) do
                local t = ls:FindFirstChild(name)
                if t and t:IsA("ValueBase") then
                    return "LS:" .. tostring(t.Value)
                end
            end
        end
        return nil
    end

    local function isTeammate(player)
        if not AimbotSettings.TeamCheck then return false end
        if player == LocalPlayer then return true end
        local myTeam = getPlayerTeam(LocalPlayer)
        local otherTeam = getPlayerTeam(player)
        if not myTeam or not otherTeam then return false end
        return myTeam == otherTeam
    end

    -- Aim Part
    local function getAimPart(character)
        if not character then return nil end
        local part = character:FindFirstChild(AimbotSettings.AimPart)
        if part and part:IsA("BasePart") then return part end
        for _, name in ipairs({"Head", "Neck", "UpperTorso", "Torso", "HumanoidRootPart"}) do
            local p = character:FindFirstChild(name)
            if p and p:IsA("BasePart") then return p end
        end
        return nil
    end

    -- Get Target
    local function getTarget()
        local camera = workspace.CurrentCamera
        if not camera then return nil end
        local center = Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y / 2)
        local closest, dist = nil, math.huge

        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and not isTeammate(p) and p.Character then
                local part = getAimPart(p.Character)
                if part then
                    local pos, onScreen = camera:WorldToViewportPoint(part.Position)
                    if onScreen then
                        local mag = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                        if mag <= Circle.Radius and mag < dist then
                            closest = part
                            dist = mag
                        end
                    end
                end
            end
        end
        return closest
    end

    -- Main Loop
    AimbotSettings.Connection = RunService.RenderStepped:Connect(function()
        local camera = workspace.CurrentCamera
        if not camera then return end

        updateCircle()
        Circle.Visible = AimbotSettings.Enabled

        if AimbotSettings.Enabled then
            local target = getTarget()
            if target then
                camera.CFrame = CFrame.lookAt(camera.CFrame.Position, target.Position)
            end
        end
    end)
end

local HitboxSettings = {
    Enabled = false,
    Size = 12,
    Transparency = 0.6,
    Color = Color3.fromRGB(160, 0, 255),
    OriginalSizes = setmetatable({}, {__mode = "k"}),
    OriginalProps = setmetatable({}, {__mode = "k"}),
    Connection = nil,
}

local function applyHitbox(player)
    local char = player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    if not HitboxSettings.OriginalSizes[hrp] then
        HitboxSettings.OriginalSizes[hrp] = hrp.Size
        HitboxSettings.OriginalProps[hrp] = {
            Transparency = hrp.Transparency,
            Material = hrp.Material,
            Color = hrp.Color,
            CanCollide = hrp.CanCollide,
        }
    end
    
    hrp.Size = Vector3.new(HitboxSettings.Size, HitboxSettings.Size, HitboxSettings.Size)
    hrp.Transparency = HitboxSettings.Transparency
    hrp.Material = Enum.Material.Neon
    hrp.Color = HitboxSettings.Color
    hrp.CanCollide = false
end

local function resetHitbox(player)
    local char = player and player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    local origSize = HitboxSettings.OriginalSizes[hrp]
    local origProps = HitboxSettings.OriginalProps[hrp]
    
    if origSize then hrp.Size = origSize end
    if origProps then
        hrp.Transparency = origProps.Transparency
        hrp.Material = origProps.Material
        hrp.Color = origProps.Color
        hrp.CanCollide = origProps.CanCollide
    end
    
    HitboxSettings.OriginalSizes[hrp] = nil
    HitboxSettings.OriginalProps[hrp] = nil
end

local function resetAllHitboxes()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then resetHitbox(p) end
    end
end

local function updateHitboxes()
    if not HitboxSettings.Enabled then return end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            applyHitbox(p)
        end
    end
end

local function setHitboxEnabled(value)
    HitboxSettings.Enabled = value and true or false
    
    if HitboxSettings.Connection then
        HitboxSettings.Connection:Disconnect()
        HitboxSettings.Connection = nil
    end
    
    if HitboxSettings.Enabled then
        HitboxSettings.Connection = RunService.RenderStepped:Connect(updateHitboxes)
    else
        resetAllHitboxes()
    end
end

local PingLabel = StatsSection:Paragraph({ Text = "Ping: Calculating..." })
local MemLabel = StatsSection:Paragraph({ Text = "Memory: Calculating..." })

track(RunService.RenderStepped:Connect(function(dt)
    local now = tick()
    if now - InfoSettings.LastUpdate < 0.5 then return end
    InfoSettings.LastUpdate = now
    pcall(function()
        FpsLabel:Set("FPS: " .. math.floor(1 / math.max(dt, 0.001)))
PingLabel:Set("Ping: " .. math.floor(LocalPlayer:GetNetworkPing() * 1000) .. " ms")
MemLabel:Set("Memory: " .. math.floor(game:GetService("Stats"):GetTotalMemoryUsageMb()) .. " MB")
    end)
end))
end

        pcall(function() setclipboard(info) end)
    end

    Text = "Copy Discord Link",
    Callback = function()

        pcall(function() setclipboard(discordLink) end)
    end
})

ActionSection:Button({
    Text = "Rejoin Server",
    Callback = function()
        pcall(function()
            game:GetService("TeleportService"):TeleportToPlaceInstance(
                game.PlaceId, game.JobId, LocalPlayer)
        end)
    end
})

ActionSection:Button({
    Text = "Server Hop",
    Callback = function()
        pcall(function()
            local url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
            local req = (syn and syn.request) or request or http_request
            if req then
                local res = req({Url = url, Method = "GET"})
                local data = game:GetService("HttpService"):JSONDecode(res.Body)
                if data and data.data then
                    for _, server in ipairs(data.data) do
                        if server.playing < server.maxPlayers and server.id ~= game.JobId then
                            game:GetService("TeleportService"):TeleportToPlaceInstance(
                                game.PlaceId, server.id, LocalPlayer)
                            break
                        end
                    end
                end
            end
        end)
    end
})
end

-- ===== KEYBINDS =====

track(LocalPlayer.CharacterAdded:Connect(function(character)
        bindMovementCharacter(character)
        if SpinSettings.Enabled then
                task.defer(function()
                        task.wait(0.5)
                        if SpinSettings.Enabled then startSpin() end
                end)
        end
end))

track(UserInputService.JumpRequest:Connect(function()
        if not running or not MovementSettings.InfiniteJump or UserInputService:GetFocusedTextBox() then return end
        local humanoid = MovementSettings.Humanoid
        if humanoid and humanoid.Parent and humanoid.Health > 0 and humanoid:GetState() ~= Enum.HumanoidStateType.Seated then
                humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
        end
end))

track(UserInputService.InputBegan:Connect(function(input, gameProcessedEvent)
        if PlatformSettings.Enabled and not UserInputService:GetFocusedTextBox() then
                if input.KeyCode == Enum.KeyCode.E then PlatformSettings.UpHeld = true; return end
                if input.KeyCode == Enum.KeyCode.Q then PlatformSettings.DownHeld = true; return end
                if input.KeyCode == Enum.KeyCode.LeftShift or input.KeyCode == Enum.KeyCode.RightShift then PlatformSettings.ForwardHeld = true; return end
        end
        if input.KeyCode == Enum.KeyCode.K and not UserInputService:GetFocusedTextBox() then Window:Toggle(); return end
end))

track(UserInputService.InputEnded:Connect(function(input)
        if input.KeyCode == Enum.KeyCode.E then PlatformSettings.UpHeld = false; return end
        if input.KeyCode == Enum.KeyCode.Q then PlatformSettings.DownHeld = false; return end
        if input.KeyCode == Enum.KeyCode.LeftShift or input.KeyCode == Enum.KeyCode.RightShift then PlatformSettings.ForwardHeld = false; return end
end))

track(Mouse.Button1Down:Connect(function()
        if not running or UserInputService:GetFocusedTextBox() then return end
        if PartRingSettings.TargetPickerEnabled then return end
        if TeleportCursorSettings.Enabled then teleportToCursor(); return end
end))

track(Players.PlayerRemoving:Connect(function(player)
        removeESP(player.Character)
end))

track(RunService.Heartbeat:Connect(function(deltaTime)
        if not running then return end
        if MovementSettings.Enabled then applySpeed() end
        if MovementSettings.JumpEnabled then applyJump() end
end))

track(Window.Gui.Destroying:Connect(cleanup))
